// Modern ES6 JavaScript Showcase
import { EventEmitter } from 'events';

class Greeter extends EventEmitter {
    constructor(greeting = 'Hello') {
        super();
        this.greeting = greeting;
    }

    greet(name) {
        const message = `${this.greeting}, ${name}!`;
        this.emit('greet', message);
        return message;
    }
}

const greeter = new Greeter('Welcome');
greeter.on('greet', (msg) => {
    console.log(`[EVENT] ${msg}`);
});

greeter.greet('World');

# Modern Python 3 Showcase
import math
from typing import List

class Vector:
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y

    def magnitude(self) -> float:
        return math.sqrt(self.x ** 2 + self.y ** 2)

    def scale(self, factor: float) -> 'Vector':
        return Vector(self.x * factor, self.y * factor)

    def __repr__(self) -> str:
        return f"Vector({self.x}, {self.y})"

# Calculate hypotenuse magnitude
v = Vector(3.0, 4.0)
print(f"Vector: {v}")
print(f"Magnitude: {v.magnitude()}")

-- Retrieve active system errors grouped by server
SELECT 
    server_id,
    COUNT(error_id) AS total_errors,
    AVG(latency_ms) AS avg_latency
FROM 
    system_logs
WHERE 
    status = 'Error'
    AND timestamp >= DATEADD(hh, -24, GETDATE())
GROUP BY 
    server_id
HAVING 
    COUNT(error_id) > 10
ORDER BY 
    total_errors DESC;
